    Add ListEACITypes handler function to internal/api/handlers/ea_handlers.go:

    ```go
    // ListEACITypes returns all EA-specific CI types
    // @Summary List EA CI Types
    // @Tags ea
    // @Accept json
    // @Produce json
    // @Success 200 {array} ci.CIType
    // @Router /ea/ci-types [get]
    func (h *EAHandlers) ListEACITypes(w http.ResponseWriter, r *http.Request) {
        ctx := r.Context()

        ciTypes, err := h.service.ListEACITypes(ctx)
        if err != nil {
            httperror.RespondWithError(w, http.StatusInternalServerError, "Failed to list EA CI types", err)
            return
        }

        respondWithJSON(w, http.StatusOK, ciTypes)
    }
    ```
